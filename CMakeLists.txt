cmake_minimum_required(VERSION 3.16)
project(flashstore_ab_partition C)

add_library(flash_store src/flash_store.c)
target_include_directories(flash_store PUBLIC include)
target_compile_features(flash_store PUBLIC c_std_11)

option(FLASH_STORE_BUILD_TESTS "Build host-side tests" ON)
if(FLASH_STORE_BUILD_TESTS)
    enable_testing()
    add_executable(flash_store_tests tests/test_flash_store.c)
    target_link_libraries(flash_store_tests PRIVATE flash_store)
    add_library(air001_example_compile OBJECT examples/air001_example.c)
    target_include_directories(air001_example_compile PRIVATE include tests/stubs)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(flash_store_tests PRIVATE -Wall -Wextra -Werror)
        target_compile_options(air001_example_compile PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME flash_store_tests COMMAND flash_store_tests)
endif()
