cmake_minimum_required(VERSION 3.16)
project(flashstore_ab_partition C)

add_library(xxtea src/xxtea.c)
target_include_directories(xxtea PUBLIC include)
target_compile_features(xxtea PUBLIC c_std_11)

add_library(chacha20 src/chacha20.c)
target_include_directories(chacha20 PUBLIC include)
target_compile_features(chacha20 PUBLIC c_std_11)

add_library(flash_store src/flash_store.c)
target_include_directories(flash_store PUBLIC include)
target_compile_features(flash_store PUBLIC c_std_11)

option(FLASH_STORE_BUILD_TESTS "Build host-side tests" ON)
if(FLASH_STORE_BUILD_TESTS)
    enable_testing()
    add_executable(flash_store_tests tests/test_flash_store.c)
    target_link_libraries(flash_store_tests PRIVATE flash_store)
    add_executable(chacha20_tests tests/test_chacha20.c)
    target_link_libraries(chacha20_tests PRIVATE chacha20)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(flash_store_tests PRIVATE -Wall -Wextra -Werror)
        target_compile_options(chacha20_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME flash_store_tests COMMAND flash_store_tests)
    add_test(NAME chacha20_tests COMMAND chacha20_tests)
endif()
